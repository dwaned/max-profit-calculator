3   import jakarta.servlet.FilterChain;
4   import jakarta.servlet.ServletException;
5   import jakarta.servlet.http.HttpServletRequest;
6   import jakarta.servlet.http.HttpServletResponse;
7   import org.slf4j.Logger;
8   import org.slf4j.LoggerFactory;
9   import org.springframework.http.HttpStatus;
10  import org.springframework.http.MediaType;
11  import org.springframework.stereotype.Component;
12  import org.springframework.web.filter.OncePerRequestFilter;
13  
14  import java.io.IOException;
15  
16  /**
17   * Servlet filter that enforces per-IP rate limiting on the {@code /api/calculate}
18   * endpoint via {@link RateLimiterService}.
19   *
20   * <p>When the bucket has tokens, the request is forwarded to the controller
21   * chain. When the bucket is empty, the filter responds directly with HTTP 429
22   * and a JSON body of the form {@code {"message": "Rate limit exceeded"}}.
23   *
24   * <p>The filter checks the request URI internally so only {@code /api/calculate}
25   * is rate-limited; other endpoints (notably {@code /api/health}) are passed
26   * through unconditionally. Client identification prefers {@code X-Forwarded-For}
27   * (Render / load balancer proxy) over {@code request.getRemoteAddr()}.
28   */
29  @Component
30  public class RateLimitFilter extends OncePerRequestFilter {
31  
32      private static final Logger LOGGER = LoggerFactory.getLogger(RateLimitFilter.class);
33      private static final String X_FORWARDED_FOR = "X-Forwarded-For";
34      private static final String CALCULATE_PATH = "/api/calculate";
35  
36      private final RateLimiterService limiter;
37  
38      public RateLimitFilter(final RateLimiterService limiter) {
39          this.limiter = limiter;
40      }
41  
42      @Override
43      protected void doFilterInternal(final HttpServletRequest request,
44                                      final HttpServletResponse response,
45                                      final FilterChain chain) throws ServletException, IOException {
46          if (!isRateLimited(request)) {
47              chain.doFilter(request, response);
48              return;
49          }
50          final String clientKey = clientKey(request);
51          if (limiter.tryAcquire(clientKey)) {
52              chain.doFilter(request, response);
53          } else {
54              LOGGER.warn("Rate limit exceeded for client {}", clientKey);
55              response.setStatus(HttpStatus.TOO_MANY_REQUESTS.value());
56              response.setContentType(MediaType.APPLICATION_JSON_VALUE);
57              response.getWriter().write("{\"message\": \"Rate limit exceeded\"}");
58          }
59      }
60  
61      /**
62       * @return {@code true} iff the request URI is the {@code /api/calculate}
63       *         endpoint and should be rate-limited. Uses {@code endsWith} so the
64       *         filter works whether or not a servlet context path is configured
65       *         (production has context path {@code /api}; MockMvc tests do not).
66       */
67      static boolean isRateLimited(final HttpServletRequest request) {
68          String uri = request.getRequestURI();
69          return uri != null && uri.endsWith("/calculate");
70      }
71  
72      /**
73       * Resolves the per-request client key. Honours {@code X-Forwarded-For} so
74       * the filter behaves correctly behind Render's reverse proxy / load balancer.
75       *
76       * @param request the inbound servlet request
77       * @return the client IP (or proxy-resolved address)
78       */
79      static String clientKey(final HttpServletRequest request) {
80          String forwarded = request.getHeader(X_FORWARDED_FOR);
81          if (forwarded != null && !forwarded.isBlank()) {
82              int comma = forwarded.indexOf(',');
83              return (comma >= 0 ? forwarded.substring(0, comma) : forwarded).trim();
84          }
85          return request.getRemoteAddr();
86      }
87  }
