32  /**
33   * REST controller exposing the {@code /api/calculate} and
34   * {@code /api/health} endpoints. {@code /api/calculate} is rate-limited
35   * via {@link com.maxprofit.calculator.controller.RateLimitFilter}; the
36   * filter is registered in {@code RateLimitFilterConfig} for the
37   * {@code /api/calculate} URL pattern only.
38   *
39   * <p>Instruments Micrometer counters/timer so a Prometheus scrape
40   * ({@code /actuator/prometheus}) exposes invocation count, execution
41   * time, and 429-rejection count.
42   *
43   * @author dwaned
44   */
45  @SuppressWarnings({"checkstyle:JavadocPackage", "checkstyle:LineLength"})
46  @RestController
47  @CrossOrigin
48  @Tag(name = "Calculator", description = "API for calculating maximum profit from stock prices")
49  public class CalculatorController {
50  
51      private final Counter invocationsCounter;
52      private final Timer executionTimer;
53  
54      public CalculatorController(final MeterRegistry meterRegistry) {
55          this.invocationsCounter = meterRegistry.counter("calculate_invocations_total");
56          this.executionTimer = meterRegistry.timer("calculate_execution_seconds");
57      }

59      /**
60       * Calculates the maximum profit that can be made from a given set of stock
61       * prices and savings amount.
62       *
63       * @param request the calculation request containing the savings amount,
64       *                current prices, and future prices
65       * @return the calculation result containing the maximum profit and the
66       * indices of the stock prices that should be bought and sold
67       * @throws IllegalArgumentException if the request is invalid (e.g., savings
68       *                                  amount is negative, prices are not in the correct range)
69       */
70      @Operation(
71              summary = "Calculate maximum profit",
72              description = "Calculates the maximum profit that can be made from a given set of stock "
73                      + "prices and savings amount. Returns the optimal buy and sell indices.")
74      @ApiResponses(value = {
75              @ApiResponse(
76                      responseCode = "200",
77                      description = "Successfully calculated maximum profit",
78                      content = @Content(mediaType = "application/json",
79                              schema = @Schema(implementation = CalculationResult.class))),
80              @ApiResponse(
81                      responseCode = "400",
82                      description = "Invalid request parameters",
83                      content = @Content(mediaType = "application/json",
84                              examples = @ExampleObject(value = "{\"error\": \"Invalid input: ...\"}"))),
85              @ApiResponse(
86                      responseCode = "500",
87                      description = "Internal server error")
88      })
89      @PostMapping(value = "/calculate", consumes = MediaType.APPLICATION_JSON_VALUE,
90              produces = MediaType.APPLICATION_JSON_VALUE)
91      @ResponseStatus(org.springframework.http.HttpStatus.OK)
92      public CalculationResult calculate(
93              @Valid @RequestBody final CalculationRequest request) {
94          invocationsCounter.increment();
95          Timer.Sample sample = Timer.start();
96          try {
97              List<String> companyNames = request.getCompanyNames();
98              if (companyNames == null || companyNames.isEmpty()) {
99                  companyNames = CompanyNameGenerator.generateCompanyNames(
100                         Math.max(request.getBuyPrices().size(), request.getSellPrices().size()));
101             }
102             return Stock.returnIndicesMaxProfit(request.getSavings(),
103                     request.getBuyPrices(), request.getSellPrices(), companyNames);
104         } catch (IllegalArgumentException e) {
105             throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "Invalid input: " + e.getMessage());
106         } finally {
107             sample.stop(executionTimer);
108         }
109     }
