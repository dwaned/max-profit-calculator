4   import jakarta.validation.constraints.Min;
5   import jakarta.validation.constraints.NotEmpty;
6   import jakarta.validation.constraints.NotNull;
7   import jakarta.validation.constraints.Size;

9   import java.util.List;

11  /**
12   * Inbound request body for {@code POST /api/calculate}. Carries the savings
13   * budget, the buy-price and sell-price lists, and an optional list of
14   * company names to display in the response.
15   *
16   * <p>Bean-validation constraints are enforced by the controller's
17   * {@code @Valid} annotation; see {@code application.properties} for the
18   * corresponding error messages.
19   *
20   * @author dwaned
21   */
22  @SuppressWarnings({"checkstyle:JavadocVariable", "checkstyle:DesignForExtension",
23      "checkstyle:FinalParameters", "checkstyle:HiddenField", "checkstyle:MagicNumber"})
24  public class CalculationRequest {
25      private static final int MAX_PRICE_LIST_SIZE = 100;
26  
27      @NotNull(message = "Savings amount is required")
28      @Min(value = 1, message = "Savings must be at least 1")
29      @Max(value = 1000, message = "Savings must not exceed 1000")
30      private int savings;
31  
32      @NotNull(message = "Buy prices are required")
33      @NotEmpty(message = "Buy prices cannot be empty")
34      @Size(max = MAX_PRICE_LIST_SIZE, message = "Buy prices must not exceed 100 entries")
35      private List<Integer> buyPrices;
36  
37      @NotNull(message = "Sell prices are required")
38      @NotEmpty(message = "Sell prices cannot be empty")
39      @Size(max = MAX_PRICE_LIST_SIZE, message = "Sell prices must not exceed 100 entries")
40      private List<Integer> sellPrices;
41  
42      private List<String> companyNames;
43  
44      public int getSavings() {
45          return savings;
46      }
47  
48      public void setSavings(int savings) {
49          this.savings = savings;
50      }
51  
52      public List<Integer> getBuyPrices() {
53          return buyPrices;
54      }
55  
56      public void setBuyPrices(List<Integer> buyPrices) {
57          this.buyPrices = buyPrices;
58      }
59  
60      public List<Integer> getSellPrices() {
61          return sellPrices;
62      }
63  
64      public void setSellPrices(List<Integer> sellPrices) {
65          this.sellPrices = sellPrices;
66      }
67  
68      public List<String> getCompanyNames() {
69          return companyNames;
70      }
71  
72      public void setCompanyNames(List<String> companyNames) {
73          this.companyNames = companyNames;
74      }
75  }
