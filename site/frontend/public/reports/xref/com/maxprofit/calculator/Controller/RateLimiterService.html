1   package com.maxprofit.calculator.controller;
2   
3   import io.github.bucket4j.Bandwidth;
4   import io.github.bucket4j.Bucket;
5   
6   import java.time.Duration;
7   import java.util.concurrent.ConcurrentHashMap;
8   import java.util.concurrent.ConcurrentMap;
9   
10  /**
11   * Per-key token-bucket rate limiter backed by Bucket4j.
12   *
13   * <p>Each key (typically a client IP) gets its own {@link Bucket} created on
14   * first use and cached for the lifetime of the service. When
15   * {@link RateLimitProperties#enabled()} is {@code false} {@link #tryAcquire}
16   * always returns {@code true}.
17   */
18  public class RateLimiterService {
19  
20      private final RateLimitProperties properties;
21      private final ConcurrentMap<String, Bucket> buckets = new ConcurrentHashMap<>();
22  
23      public RateLimiterService(final RateLimitProperties properties) {
24          this.properties = properties;
25      }
26  
27      /**
28       * Attempts to consume one token from the bucket associated with {@code key}.
29       *
30       * @param key the client identifier (typically a remote IP)
31       * @return {@code true} if the request may proceed; {@code false} if it
32       *         should be rejected with HTTP 429
33       */
34      public boolean tryAcquire(final String key) {
35          if (!properties.enabled()) {
36              return true;
37          }
38          Bucket bucket = buckets.computeIfAbsent(key, k -> newBucket());
39          return bucket.tryConsume(1);
40      }
41  
42      private Bucket newBucket() {
43          Bandwidth limit = Bandwidth.builder()
44                  .capacity(properties.capacity())
45                  .refillGreedy(properties.refillTokens(),
46                          Duration.ofSeconds(properties.refillPeriodSeconds()))
47                  .build();
48          return Bucket.builder().addLimit(limit).build();
49      }
50  }
