1   package com.maxprofit.calculator.controller;

3   import org.springframework.http.HttpStatus;
4   import org.springframework.http.HttpStatusCode;
5   import org.springframework.http.ResponseEntity;
6   import org.springframework.http.converter.HttpMessageNotReadableException;
7   import org.springframework.web.bind.MethodArgumentNotValidException;
8   import org.springframework.web.bind.annotation.ExceptionHandler;
9   import org.springframework.web.bind.annotation.RestControllerAdvice;
10  import org.springframework.web.server.ResponseStatusException;
11  
12  import java.util.Map;
13  
14  /**
15   * Maps request errors to a single JSON shape, {@code {"message": "..."}},
16   * which the frontend displays and the Pact contract verifies.
17   */
18  @RestControllerAdvice
19  public class GlobalExceptionHandler {
20  
21      @ExceptionHandler(MethodArgumentNotValidException.class)
22      public ResponseEntity<Map<String, String>> handleValidationException(MethodArgumentNotValidException ex) {
23          String message = ex.getBindingResult().getFieldErrors().stream()
24                  .findFirst()
25                  .map(error -> "Invalid input: " + error.getDefaultMessage())
26                  .orElse("Validation failed");
27          return body(HttpStatus.BAD_REQUEST, message);
28      }
29  
30      @ExceptionHandler(HttpMessageNotReadableException.class)
31      public ResponseEntity<Map<String, String>> handleUnreadableBody(HttpMessageNotReadableException ex) {
32          return body(HttpStatus.BAD_REQUEST, "Invalid input: malformed request body");
33      }
34  
35      @ExceptionHandler(ResponseStatusException.class)
36      public ResponseEntity<Map<String, String>> handleResponseStatus(ResponseStatusException ex) {
37          String reason = ex.getReason() != null ? ex.getReason() : "Request failed";
38          return body(ex.getStatusCode(), reason);
39      }
40  
41      private static ResponseEntity<Map<String, String>> body(HttpStatusCode status, String message) {
42          return ResponseEntity.status(status).body(Map.of("message", message));
43      }
44  }
