1   package com.maxprofit.calculator;

3   import org.junit.jupiter.api.Test;
4   
5   import java.util.Collections;
6   import java.util.List;
7   
8   import static org.junit.jupiter.api.Assertions.assertEquals;
9   import static org.junit.jupiter.api.Assertions.assertNotNull;
10  
11  /**
12   * Tests for {@link CalculationResult} constructor invariants.
13   *
14   * Specifically guards against {@code getCompanyNames()} returning {@code null}
15   * for callers that use the no-arg or 4-arg constructors — every caller
16   * should be able to iterate {@code getCompanyNames()} without a null check.
17   */
18  @SuppressWarnings({"checkstyle:magicnumber", "checkstyle:LineLength"})
19  class CalculationResultTests {
20  
21      @Test
22      void fourArgConstructorShouldReturnNonNullCompanyNames() {
23          CalculationResult result = new CalculationResult(0, Collections.emptyList(), 0, 0);
24          assertNotNull(result.getCompanyNames(), "getCompanyNames() must not return null");
25      }
26  
27      @Test
28      void fourArgConstructorShouldReturnEmptyCompanyNamesWhenNotProvided() {
29          CalculationResult result = new CalculationResult(0, Collections.emptyList(), 0, 0);
30          List<String> names = result.getCompanyNames();
31          assertEquals(0, names.size(), "getCompanyNames() should be empty by default");
32      }
33  
34      @Test
35      void noArgConstructorShouldReturnNonNullCompanyNames() {
36          CalculationResult result = new CalculationResult();
37          assertNotNull(result.getCompanyNames(), "getCompanyNames() must not return null");
38      }
39  
40      @Test
41      void fiveArgConstructorShouldStillReturnProvidedCompanyNames() {
42          List<String> provided = List.of("Acme", "Beta");
43          CalculationResult result = new CalculationResult(0, Collections.emptyList(), 0, 0, provided);
44          assertEquals(provided, result.getCompanyNames(),
45                  "Explicit company names must be preserved verbatim");
46      }
47  }
