17  /**
18   * End-to-end checks, on a real embedded Tomcat, for CORS, forwarded-header
19   * handling in the rate limiter, input bounds and the error response shape.
20   */
21  @SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT, properties = {
22          "app.ratelimit.capacity=3",
23          "app.ratelimit.refill-tokens=1",
24          "app.ratelimit.refill-period-seconds=3600",
25          "app.cors.allowed-origins=https://allowed.example.com"
26  })
27  class ApiSecurityTest {
28  
29      private static final String VALID_BODY =
30              "{\"savings\":10,\"buyPrices\":[5,5,10],\"sellPrices\":[15,10,35]}";
31  
32      // TestRestTemplate already prefixes the /api servlet context path.
33      @Autowired
34      private TestRestTemplate rest;
35  
36      private ResponseEntity<String> postCalculate(final String body, final HttpHeaders extra) {
37          HttpHeaders headers = new HttpHeaders();
38          headers.setContentType(MediaType.APPLICATION_JSON);
39          headers.addAll(extra);
40          return rest.postForEntity("/calculate", new HttpEntity<>(body, headers), String.class);
41      }
42  
43      private ResponseEntity<String> preflight(final String origin) {
44          HttpHeaders headers = new HttpHeaders();
45          headers.setOrigin(origin);
46          headers.setAccessControlRequestMethod(HttpMethod.POST);
47          return rest.exchange("/calculate", HttpMethod.OPTIONS, new HttpEntity<>(headers), String.class);
48      }
49  
50      @Test
51      @DisplayName("CORS preflight succeeds for a configured origin")
52      void preflightAllowedOrigin() {
53          ResponseEntity<String> response = preflight("https://allowed.example.com");
54          assertThat(response.getStatusCode()).isEqualTo(HttpStatus.OK);
55          assertThat(response.getHeaders().getAccessControlAllowOrigin())
56                  .isEqualTo("https://allowed.example.com");
57      }
58  
59      @Test
60      @DisplayName("CORS preflight is rejected for an unknown origin")
61      void preflightUnknownOrigin() {
62          ResponseEntity<String> response = preflight("https://evil.example.com");
63          assertThat(response.getStatusCode()).isEqualTo(HttpStatus.FORBIDDEN);
64          assertThat(response.getHeaders().getAccessControlAllowOrigin()).isNull();
65      }
66  
67      @Test
68      @DisplayName("Forged X-Forwarded-For entries do not give a client a fresh bucket")
69      void forgedForwardedForDoesNotBypassRateLimit() {
70          // Requests arrive from loopback, a trusted proxy, carrying the client IP
71          // in CF-Connecting-IP (as Cloudflare/nginx set it). X-Forwarded-For is
72          // ignored, so rotating it must not reset the client's bucket.
73          for (int i = 0; i < 3; i++) {
74              assertThat(postCalculate(VALID_BODY, client("198.51.100.7", "203.0.113." + i))
75                      .getStatusCode()).isEqualTo(HttpStatus.OK);
76          }
77          assertThat(postCalculate(VALID_BODY, client("198.51.100.7", "203.0.113.200"))
78                  .getStatusCode()).isEqualTo(HttpStatus.TOO_MANY_REQUESTS);
79  
80          // A different client IP still has its own bucket.
81          assertThat(postCalculate(VALID_BODY, client("198.51.100.8", "203.0.113.201"))
82                  .getStatusCode()).isEqualTo(HttpStatus.OK);
83      }
84  
85      private static HttpHeaders client(final String clientIp, final String forgedForwardedFor) {
86          HttpHeaders headers = new HttpHeaders();
87          headers.add("CF-Connecting-IP", clientIp);
88          headers.add("X-Forwarded-For", forgedForwardedFor);
89          return headers;
90      }
91  
92      @Test
93      @DisplayName("Prices above the maximum are rejected with a message body")
94      void priceAboveMaximumRejected() {
95          HttpHeaders headers = new HttpHeaders();
96          headers.add("CF-Connecting-IP", "198.51.100.20");
97          ResponseEntity<String> response = postCalculate(
98                  "{\"savings\":10,\"buyPrices\":[5],\"sellPrices\":[2000000000]}", headers);
99          assertThat(response.getStatusCode()).isEqualTo(HttpStatus.BAD_REQUEST);
100         assertThat(response.getBody()).contains("\"message\":\"Invalid input: Sell prices must not exceed 1000\"");
101     }
102 
103     @Test
104     @DisplayName("A null price is rejected with 400, not a 500")
105     void nullPriceRejected() {
106         HttpHeaders headers = new HttpHeaders();
107         headers.add("CF-Connecting-IP", "198.51.100.21");
108         ResponseEntity<String> response = postCalculate(
109                 "{\"savings\":10,\"buyPrices\":[5,null],\"sellPrices\":[6,7]}", headers);
110         assertThat(response.getStatusCode()).isEqualTo(HttpStatus.BAD_REQUEST);
111         assertThat(response.getBody()).contains("Buy prices must not contain empty values");
112     }

125     @Test
126     @DisplayName("Mismatched list sizes return the real reason in the message body")
127     void mismatchedSizesReturnMessage() {
128         HttpHeaders headers = new HttpHeaders();
129         headers.add("CF-Connecting-IP", "198.51.100.23");
130         ResponseEntity<String> response = postCalculate(
131                 "{\"savings\":10,\"buyPrices\":[5,6],\"sellPrices\":[7]}", headers);
132         assertThat(response.getStatusCode()).isEqualTo(HttpStatus.BAD_REQUEST);
133         assertThat(response.getBody()).contains("\"message\":\"Invalid input:").contains("same size");
134     }

136     @Test
137     @DisplayName("Malformed JSON returns a message body")
138     void malformedJsonReturnsMessage() {
139         HttpHeaders headers = new HttpHeaders();
140         headers.add("CF-Connecting-IP", "198.51.100.24");
141         ResponseEntity<String> response = postCalculate("{not json", headers);
142         assertThat(response.getStatusCode()).isEqualTo(HttpStatus.BAD_REQUEST);
143         assertThat(response.getBody()).isEqualTo("{\"message\":\"Invalid input: malformed request body\"}");
144     }

147     @DisplayName("Actuator health does not expose component details")
148     void actuatorHealthHidesDetails() {
149         ResponseEntity<String> response = rest.getForEntity("/actuator/health", String.class);
150         assertThat(response.getStatusCode()).isEqualTo(HttpStatus.OK);
