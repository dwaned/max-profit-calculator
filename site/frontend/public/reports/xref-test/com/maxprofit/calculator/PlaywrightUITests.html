12  /**
13   * End-to-end browser test for the calculator UI.
14   *
15   * <p>Drives the real frontend (not the API) through a real Chromium instance
16   * to confirm the form, the API call, and the results card all render and
17   * work together. Runs under the {@code -Pplaywright-tests} profile which
18   * adds the {@code com.microsoft.playwright:playwright} dependency and gates
19   * this class behind a profile-scoped surefire {@code <include>}.
20   *
21   * <p>The test expects the frontend to be reachable at the URL below
22   * ({@code PLAYWRIGHT_BASE_URL}, defaults to {@code http://localhost:3000}).
23   * In CI the {@code containers.yml} job brings the full stack up via
24   * docker-compose before running {@code mvn test -Pplaywright-tests}.
25   *
26   * <p>The default surefire config in {@code pom.xml} excludes this class so
27   * it does not run under {@code mvn test} (where there is no browser binary
28   * and no frontend). The reports workflow likewise excludes it explicitly
29   * via {@code -Dtest='…,!PlaywrightUITests, …'}.
30   */
31  @SuppressWarnings({"checkstyle:LineLength", "checkstyle:magicnumber"})
32  public class PlaywrightUITests {
33      /** Base URL where the React dev server / nginx is reachable. */
34      private static final String BASE_URL = System.getenv().getOrDefault(
35          "PLAYWRIGHT_BASE_URL", "http://localhost:3000"
36      );
37  
38      /** Browser object. */
39      private Browser browser;
40      /** Page object. */
41      private Page page;
42  
43      /**
44       * Setup the browser and page.
45       */
46      @BeforeEach
47      public void setUp() {
48          browser = Playwright.create().chromium().launch(
49              new BrowserType.LaunchOptions().setHeadless(true)
50          );
51          page = browser.newPage();
52      }
53  
54      /**
55       * Close the browser.
56       */
57      @AfterEach
58      public void tearDown() {
59          browser.close();
60      }
61  
62      /**
63       * Drive the calculator form with the default sample data
64       * (savings=10, buy=[5,5,10], sell=[15,10,35]) and assert the
65       * algorithm picks stock #2 for max profit of 25.
66       */
67      @Test
68      public void testCalculateUsingUI() {
69          page.navigate(BASE_URL + "/#/calculator");
70  
71          // Wait for the form to render before touching it. The dev server /
72          // first load can take a few seconds in CI.
73          page.waitForSelector("#savings-amount");
74          page.waitForSelector("button[type='submit']");
75  
76          // Fill the form (aria-labels are stable selectors across the
77          // desktop/table and mobile/card layouts).
78          page.fill("#savings-amount", "10");
79          page.locator("input[aria-label='Buy price for stock 1']").fill("5");
80          page.locator("input[aria-label='Buy price for stock 2']").fill("5");
81          page.locator("input[aria-label='Buy price for stock 3']").fill("10");
82          page.locator("input[aria-label='Sell price for stock 1']").fill("15");
83          page.locator("input[aria-label='Sell price for stock 2']").fill("10");
84          page.locator("input[aria-label='Sell price for stock 3']").fill("35");
85  
86          // Submit and wait for the result card to render.
87          page.click("button[type='submit']");
88          page.waitForSelector("text=Max Profit");
89  
90          // The result is rendered inside the "Max Profit" tile as
91          // `€<value>`. Stock #2 (buy=10, sell=35) is the unique max-profit
92          // pick for this input set.
93          String maxProfitText = page.locator("text=€").first().textContent();
94          Assertions.assertTrue(
95              maxProfitText.contains("25"),
96              "Expected max profit to be €25, got: " + maxProfitText
97          );
98      }
99  }
