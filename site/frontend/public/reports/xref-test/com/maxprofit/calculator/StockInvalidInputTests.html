15  /**
16   * Property-based tests asserting {@link Stock#returnIndicesMaxProfit} throws
17   * {@link IllegalArgumentException} (not silent empty results) for invalid inputs.
18   * This is the only behavior change in the v2 milestone; see HYG-04.
19   */
20  @SuppressWarnings({"checkstyle:LineLength", "checkstyle:magicnumber"})
21  class StockInvalidInputTests {
22  
23      static Stream<Arguments> invalidInputs() {
24          return Stream.of(
25                  Arguments.of("null currentValue",   null,                     Arrays.asList(1, 2, 3), "currentValue"),
26                  Arguments.of("null futureValue",    Arrays.asList(1, 2, 3),   null,                   "futureValue"),
27                  Arguments.of("mismatched sizes",    Arrays.asList(1, 2),     Arrays.asList(1),       "size"),
28                  Arguments.of("non-positive current", Arrays.asList(0, 1),     Arrays.asList(1, 2),   "positive"),
29                  Arguments.of("non-positive future",  Arrays.asList(1, 2),     Arrays.asList(1, 0),   "positive"),
30                  Arguments.of("oversize list",       Collections.nCopies(101, 1), Collections.nCopies(101, 2), "exceed")
31          );
32      }
33  
34      @ParameterizedTest
35      @MethodSource("invalidInputs")
36      void throwsOnInvalidInput(final String label,
37                                final List<Integer> currentValue,
38                                final List<Integer> futureValue,
39                                final String expectedMessageFragment) {
40          final IllegalArgumentException ex = assertThrows(
41                  IllegalArgumentException.class,
42                  () -> Stock.returnIndicesMaxProfit(10, currentValue, futureValue));
43          assertThat(ex.getMessage())
44                  .as("message for case '%s'", label)
45                  .containsIgnoringCase(expectedMessageFragment);
46      }
47  }
