17  /**
18   * Stress tests for the {@link Stock#returnIndicesMaxProfit} algorithm.
19   *
20   * <p>Pure in-process tests — no Spring context, no HTTP. They measure
21   * how the algorithm holds up under increasingly heavy inputs:
22   * <ul>
23   *   <li>Execution time at the maximum allowed input size (100 stocks)</li>
24   *   <li>Memory usage at the maximum allowed input size</li>
25   *   <li>No {@link OutOfMemoryError} at the maximum allowed input size</li>
26   * </ul>
27   *
28   * <p>These are not "performance tests" in the usual sense
29   * (response time, throughput, latency) — that is
30   * {@link ApiPerformanceTests}. Stress tests push the system past
31   * expected operating conditions to catch robustness issues (OOM,
32   * runaway allocations, runaway time).
33   *
34   * <p><b>Thresholds:</b>
35   * <ul>
36   *   <li>5 items: &lt; 10ms</li>
37   *   <li>10 items: &lt; 100ms</li>
38   *   <li>50 items: &lt; 50ms</li>
39   *   <li>100 items (boundary): &lt; 500ms</li>
40   *   <li>Memory usage: &lt; 512MB for max input</li>
41   * </ul>
42   *
43   * <p>Thresholds for 50/100 items assume the polynomial DP rewrite in Phase 2
44   * (O(n · savings)). The original brute-force permutation algorithm only
45   * passed the relaxed thresholds (500ms / 10s) because the input size cap of
46   * 100 keeps 2^100 impractical.
47   *
48   * <p><b>Running these tests:</b>
49   * <pre>
50   * mvn test -Dtest=StressTests
51   * </pre>
52   */

87      /**
88       * Tests that the algorithm completes in under 10ms for small input (5 items).
89       * 
90       * <p>This test verifies the algorithm performs efficiently for small datasets.
91       * Uses 1000 iterations with 100 warmup runs to get stable measurements.
92       */
93      @Test
94      void algorithmShouldCompleteInUnder10msFor5Items() {
95          List<Integer> currentValues = generateRandomPrices(SMALL_SIZE);
96          List<Integer> futureValues = generateRandomPrices(SMALL_SIZE);
97  
98          long executionTime = measureAlgorithmExecution(1000, 100, currentValues, futureValues);
99  
100         System.out.println("5 items - Avg execution time: " + executionTime + "ms");
101         assertTrue(executionTime < SMALL_THRESHOLD_MS,
102                 "Execution time should be < " + SMALL_THRESHOLD_MS + "ms, was: " + executionTime + "ms");
103     }

164      * Tests that memory usage doesn't exceed 512MB when processing max input.

178 
179         MemoryUsage heapAfter = memoryBean.getHeapMemoryUsage();
180         long usedAfter = heapAfter.getUsed() / (1024 * 1024);

208     /**
209      * Measures algorithm execution time with warmup and averaging.
210      * 
211      * @param totalRuns   Number of timing iterations to average
212      * @param warmupRuns  Number of warmup iterations (not timed)
213      * @param currentValues Current stock prices
214      * @param futureValues  Future stock prices
215      * @return Average execution time in milliseconds
216      */
217     private long measureAlgorithmExecution(int totalRuns, int warmupRuns, 
218             List<Integer> currentValues, List<Integer> futureValues) {
219         AtomicLong totalTime = new AtomicLong(0);
220 
221         for (int i = 0; i < warmupRuns; i++) {
222             Stock.returnIndicesMaxProfit(100, currentValues, futureValues);
223         }
224 
225         for (int i = 0; i < totalRuns; i++) {
226             long startTime = System.nanoTime();
227             Stock.returnIndicesMaxProfit(100, currentValues, futureValues);
228             long endTime = System.nanoTime();
229             totalTime.addAndGet(endTime - startTime);
230         }
231 
232         return totalTime.get() / (1_000_000 * totalRuns);
233     }

235     /**
236      * Generates a list of random stock prices.
237      * 
238      * @param size Number of prices to generate
239      * @return List of random prices between 1 and 100
240      */
241     private List<Integer> generateRandomPrices(int size) {
242         List<Integer> prices = new ArrayList<>();
243         for (int i = 0; i < size; i++) {
244             prices.add(random.nextInt(100) + 1);
245         }
246         return prices;
247     }
