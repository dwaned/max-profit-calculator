9   import static org.junit.jupiter.api.Assertions.assertEquals;
10  import static org.junit.jupiter.api.Assertions.assertTrue;
11  import static org.junit.jupiter.api.Assertions.assertFalse;
12  
13  /**
14   * Property-based tests for the {@code /api/calculate} rate limiter (SEC-05).
15   *
16   * <p>Verifies the Bucket4j-backed limiter honours the configured per-IP
17   * capacity under fuzzed burst patterns. Uses a frozen {@link RateLimiterService}
18   * constructed directly with a small capacity (5) so the property can exhaust
19   * the bucket in a single test invocation without waiting for refill.
20   *
21   * <p>Invariants checked:
22   * <ul>
23   *   <li>For any IP key, the first {@code capacity} rapid {@code tryAcquire}
24   *       calls return {@code true}.</li>
25   *   <li>The next call (the {@code capacity + 1}-th) returns {@code false}.</li>
26   *   <li>Different IP keys have independent buckets — a request from IP A
27   *       being throttled does not affect IP B.</li>
28   * </ul>
29   */
30  @SuppressWarnings({"checkstyle:LineLength", "checkstyle:magicnumber"})
31  class RateLimiterServiceTests {
32  
33      private static final int CAPACITY = 5;
34  
35      @Property
36      void firstCapacityAcquiresSucceedForAnyKey(@ForAll("ipKeys") final String key) {
37          RateLimiterService limiter = new RateLimiterService(
38                  new RateLimitProperties(true, CAPACITY, CAPACITY, 1));
39  
40          for (int i = 0; i < CAPACITY; i++) {
41              assertTrue(limiter.tryAcquire(key),
42                      "Acquires 1.." + CAPACITY + " must succeed for key=" + key);
43          }
44          assertFalse(limiter.tryAcquire(key),
45                  "Acquire " + (CAPACITY + 1) + " must be rejected for key=" + key);
46      }
47  
48      @Property
49      void keysHaveIndependentBuckets(@ForAll("ipKeys") final String keyA,
50                                      @ForAll("ipKeys") final String keyB) {
51          org.junit.jupiter.api.Assumptions.assumeTrue(!keyA.equals(keyB),
52                  "Property requires distinct keys");
53  
54          RateLimiterService limiter = new RateLimiterService(
55                  new RateLimitProperties(true, CAPACITY, CAPACITY, 1));
56  
57          // Exhaust keyA's bucket.
58          for (int i = 0; i < CAPACITY; i++) {
59              assertTrue(limiter.tryAcquire(keyA));
60          }
61          assertFalse(limiter.tryAcquire(keyA));
62  
63          // keyB's bucket should still be fresh.
64          for (int i = 0; i < CAPACITY; i++) {
65              assertTrue(limiter.tryAcquire(keyB),
66                      "keyB must have an independent bucket; got false at acquire " + (i + 1));
67          }
68          assertFalse(limiter.tryAcquire(keyB));
69      }
70  
71      @Property
72      void disabledLimiterNeverRejects(@ForAll("ipKeys") final String key) {
73          RateLimiterService limiter = new RateLimiterService(
74                  new RateLimitProperties(false, CAPACITY, CAPACITY, 1));
75  
76          // Many rapid calls — none should be rejected when the limiter is off.
77          for (int i = 0; i < CAPACITY * 3; i++) {
78              assertTrue(limiter.tryAcquire(key),
79                      "Disabled limiter must never reject; got false at " + i);
80          }
81      }
82  
83      @Property
84      void totalAcceptedCountEqualsCapacity(@ForAll("ipKeys") final String key,
85                                            @ForAll("burstSizes") final int burstSize) {
86          org.junit.jupiter.api.Assumptions.assumeTrue(burstSize >= 0);
87  
88          RateLimiterService limiter = new RateLimiterService(
89                  new RateLimitProperties(true, CAPACITY, CAPACITY, 1));
90  
91          int accepted = 0;
92          for (int i = 0; i < burstSize; i++) {
93              if (limiter.tryAcquire(key)) {
94                  accepted++;
95              }
96          }
97  
98          assertEquals(Math.min(burstSize, CAPACITY), accepted,
99                  "Accepted count must be min(burstSize, capacity)");
100     }
101 
102     @Provide
103     Arbitrary<String> ipKeys() {
104         return Arbitraries.strings().withCharRange('a', 'z').ofMinLength(1).ofMaxLength(20);
105     }
106 
107     @Provide
108     Arbitrary<Integer> burstSizes() {
109         return Arbitraries.integers().between(0, 50);
110     }
111 }
