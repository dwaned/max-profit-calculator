1   /**
2    * This class defines the step definitions for Cucumber scenarios related to
3    * stock market calculations.
4    */
5   package com.maxprofit.calculator.steps;
6   
7   import com.maxprofit.calculator.CalculationResult;
8   import com.maxprofit.calculator.Stock;
9   import io.cucumber.java.en.And;
10  import io.cucumber.java.en.Given;
11  import io.cucumber.java.en.Then;
12  import io.cucumber.java.en.When;
13  import org.slf4j.Logger;
14  import org.slf4j.LoggerFactory;
15  
16  import java.util.List;
17  import java.util.stream.Collectors;
18  import java.util.stream.Stream;
19  
20  import static org.junit.jupiter.api.Assertions.assertEquals;
21  import static org.junit.jupiter.api.Assertions.assertTrue;
22  
23  public class StepDefinitions {
24  
25      @SuppressWarnings({"checkstyle:VisibilityModifier", "checkstyle:JavadocVariable", "checkstyle:LineLength"})
26      private static final Logger LOGGER = LoggerFactory.getLogger(StepDefinitions.class);
27  
28      /**
29       * Integer storing the amount of savings.
30       */
31      private int savingsContext = 0;
32  
33      /**
34       * List storing the current prices.
35       */
36      private List<Integer> currentPricesContext;
37  
38      /**
39       * List storing the future prices.
40       */
41      private List<Integer> futurePricesContext;
42  
43      /**
44       * Sets the value of savingsContext to the given savings.
45       *
46       * @param savings an integer representing the amount of savings.
47       */
48      @Given("I have {int} Euros of savings")
49      public void iHaveEurosOfSavings(final int savings) {
50          System.out.format("Savings: %d\n", savings);
51          savingsContext = savings;
52      }
53  
54      /**
55       * Sets the value of currentPricesContext to a List of integers parsed
56       * from the given string.
57       *
58       * @param currentPrices a string representing a comma-separated list
59       *                      of current stock prices.
60       */
61      @When("Array of current stock prices are {string}")
62      public void arrayOfCurrentStockPricesAre(final String currentPrices) {
63          System.out.format("Current prices: %s\n", currentPrices);
64          currentPricesContext =
65                  Stream
66                          .of(currentPrices.split(","))
67                          .map(String::trim)
68                          .map(Integer::parseInt)
69                          .collect(Collectors.toList());
70      }
71  
72      /**
73       * Compares the expected result to the actual result of the
74       * calculation for the
75       * best combination of indices for maximum profit.
76       *
77       * @param result a string representing the expected result.
78       */
79      @Then("the best combination of indices for max profit is {string}")
80      public void theBestCombinationOfIndicesForMaxProfitIs(final String result) {
81          final List<Integer> resultIndices = Stream
82                  .of(result.split(","))
83                  .map(String::trim)
84                  .map(Integer::parseInt)
85                  .collect(Collectors.toList());
86  
87          final CalculationResult actualResult = Stock.returnIndicesMaxProfit(
88                  savingsContext,
89                  currentPricesContext,
90                  futurePricesContext
91          );
92  
93          assertEquals(
94                  resultIndices,
95                  actualResult.getIndices(),
96                  "Actual Result: "
97                          + actualResult.getIndices()
98                          + " with profit of "
99                          + actualResult.getMaxProfit()
100         );
101     }
102 
103     /**
104      * Sets the value of futurePricesContext to a List of integers parsed
105      * from the
106      * given string.
107      *
108      * @param futurePrices a string representing a comma-separated list of
109      *                     future stock prices.
110      */
111     @And("Array of future stock prices are {string}")
112     public void arrayOfFutureStockPricesAre(final String futurePrices) {
113         System.out.format("Future prices: %s\n", futurePrices);
114         futurePricesContext =
115                 Stream
116                         .of(futurePrices.split(","))
117                         .map(String::trim)
118                         .map(Integer::parseInt)
119                         .collect(Collectors.toList());
120     }
121 
122     /**
123      * Compares the expected profit to the actual profit of the calculation.
124      *
125      * @param profit an integer representing the expected profit.
126      */
127     @Then("profit is {int} Euros")
128     public void profitIsEuros(final int profit) {
129         System.out.format("Profit: %d\n", profit);
130         int actualProfit = Stock.returnIndicesMaxProfit(
131                         savingsContext,
132                         currentPricesContext,
133                         futurePricesContext
134                 )
135                 .getMaxProfit();
136         assertEquals(profit, actualProfit,
137                 "Actual Profit is %s " + actualProfit);
138     }
139 
140     /**
141      * Asserts that there is no set of indices with a profit.
142      */
143     @Then("there is no best combination for max profit")
144     public void thereIsNoBestCombinationForMaxProfit() {
145         final CalculationResult actualResult = Stock.returnIndicesMaxProfit(
146                 savingsContext,
147                 currentPricesContext,
148                 futurePricesContext
149         );
150         assertEquals(
151                 0,
152                 actualResult.getIndices().size(),
153                 "Actual Result: " + actualResult.getIndices()
154         );
155     }
156 
157     /**
158      * Asserts that profit is 0.
159      */
160     @And("no profit is made")
161     public void noProfitIsMade() {
162         final int actualProfit = Stock.returnIndicesMaxProfit(
163                         savingsContext,
164                         currentPricesContext,
165                         futurePricesContext
166                 )
167                 .getMaxProfit();
168         assertEquals(0, actualProfit, "Actual Profit is %s " + actualProfit);
169     }
170 
171     /**
172      * Asserts that with the given indices, same profit is achieved as
173      * previously stored.
174      *
175      * @param indices combination representing the indices of stocks.
176      */
177     @And("same savings and max profit is achieved with the indices {string}")
178     public void sameSavingsMaxProfitIsWithCombination(final String indices) {
179         this.theBestCombinationOfIndicesForMaxProfitIs(indices);
180     }
181 }
