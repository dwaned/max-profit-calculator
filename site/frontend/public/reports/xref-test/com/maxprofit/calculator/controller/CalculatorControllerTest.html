54      /**
55       * @throws Exception if an error occurs during the test
56       */
57      @Test
58      public void testCalculate() throws Exception {
59          CalculationRequest request = new CalculationRequest();
60          request.setSavings(10);
61          request.setBuyPrices(Arrays.asList(5, 5, 10));
62          request.setSellPrices(Arrays.asList(15, 10, 35));
63          String jsonRequest = objectMapper.writeValueAsString(request);
64  
65          CalculationResult response = Stock.returnIndicesMaxProfit(request.getSavings(), request.getBuyPrices(), request.getSellPrices());
66  
67          mockMvc.perform(post("/calculate")
68                          .contentType(MediaType.APPLICATION_JSON)
69                          .content(jsonRequest))
70                  .andExpect(status().isOk())
71                  .andExpect(content().contentType(MediaType.APPLICATION_JSON))
72                  .andExpect(jsonPath("$.maxProfit").value(response.getMaxProfit()));
73      }
